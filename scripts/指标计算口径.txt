（2）优质存款
    SELECT P1.ACCNT_ORG_ID
   ,sum((CASE WHEN P1.MNG_ACCNT_PROD_ID like 'L11002%' or P1.MNG_ACCNT_PROD_ID LIKE 'L12002%' THEN '1'
         WHEN (P1.MNG_ACCNT_PROD_ID LIKE 'L11001%' or  P1.MNG_ACCNT_PROD_ID LIKE 'L12001%') AND CAST(P1.ACCT_TERM AS DECIMAL(18,4))<='730' THEN '1'
      WHEN (P1.MNG_ACCNT_PROD_ID LIKE 'L11001%' or  P1.MNG_ACCNT_PROD_ID LIKE 'L12001%') AND CAST(P1.ACCT_TERM AS DECIMAL(18,4))>'730' AND P1.EXEC_RATE<= P4.PARA_SCOPE THEN '1'
                  ELSE '0' 
     END)*P1.CUR_BAL) as 时点余额,sum((CASE WHEN P1.MNG_ACCNT_PROD_ID like 'L11002%' or P1.MNG_ACCNT_PROD_ID LIKE 'L12002%' THEN '1'
         WHEN (P1.MNG_ACCNT_PROD_ID LIKE 'L11001%' or  P1.MNG_ACCNT_PROD_ID LIKE 'L12001%') AND CAST(P1.ACCT_TERM AS DECIMAL(18,4))<='730' THEN '1'
      WHEN (P1.MNG_ACCNT_PROD_ID LIKE 'L11001%' or  P1.MNG_ACCNT_PROD_ID LIKE 'L12001%') AND CAST(P1.ACCT_TERM AS DECIMAL(18,4))>'730' AND P1.EXEC_RATE<= P4.PARA_SCOPE THEN '1'
                  ELSE '0' 
     END)*P1.BAL_YEAR_ACCUM)/天数 as 年日均积数
    
  FROM      shenzhen_data.i01_sz_acct_sum_m P1
  LEFT JOIN shenzhen_data.FDL_REFL_DIM_PARAMET_CONFIG P4
  ON        (CASE WHEN SUBSTR(P1.MNG_ACCNT_PROD_ID,1,6) = 'L12001' THEN 'CORP_DEP_RT'
                  WHEN SUBSTR(P1.MNG_ACCNT_PROD_ID,1,6) = 'L11001' THEN 'INDIV_DEP_RT'
            ELSE '' END ) = P4.DIM_CD
  AND       P1.INT_ST_DT >= CAST(P4.PARA_ST_VAL AS DATE)
  AND       P1.INT_ST_DT <  CAST(P4.PARA_END_VAL AS DATE)
  AND       (CASE WHEN CAST(P1.ACCT_TERM AS DECIMAL(18,4)) > '730' THEN '3Y'
                  ELSE '' END ) = P4.PARA_CD 
  AND       P4.YEAR = SUBSTR('2025-12-31',1,4)
  AND       P4.BIZ_TYPE_CD = 'B_DP_RT'
  WHERE     P1.DATA_DT='2025-12-31'
  AND       P1.BASIS_CD IN('M_200')
  AND       P1.CCY_CD  IN        ('156','B02') 
  and   SUBSTR(P1.MNG_ACCNT_PROD_ID,1,4) = 'L120'
  group by P1.ACCNT_ORG_ID
  order by P1.ACCNT_ORG_ID; （分发给各机构时须卡机构ID）
